** 1.3 Models

Two-model strategy: a *coder* model for structured JSON classification/triage,
and a *general* instruct model for prose summaries that capture nuance.

With 16 GB VRAM the 5070Ti can run either model at Q5_K_M (~9 GB) cleanly.
The =MODEL_TRIAGE= and =MODEL_SUMMARY= constants in Section 2 are the only
things to change if you swap quants.

** 1.4 Reddit API Access (Optional)

The pipeline works out of the box using Reddit's public JSON endpoints (~10

def load_cache():
    """Load cache, migrate old list format if needed, prune entries >CACHE_TTL_DAYS old."""
    empty = {k: {} for k in CACHE_KEYS}
    if not os.path.exists(CACHE_FILE):
        return empty
    with open(CACHE_FILE) as f:
        data = json.load(f)
    # Ensure all keys exist
    for k in CACHE_KEYS:
        if k not in data:
            data[k] = {}
    # Migrate old list-of-strings format → dict-with-dates
    today = datetime.date.today().isoformat()
    for k in CACHE_KEYS:
        if isinstance(data[k], list):
            data[k] = {item: today for item in data[k]}
    # Prune entries older than TTL
    cutoff = (datetime.date.today() - datetime.timedelta(days=CACHE_TTL_DAYS)).isoformat()
    pruned = {k: {id_: d for id_, d in v.items() if d >= cutoff} for k, v in data.items()}
    return pruned

def save_cache(c):
    with open(CACHE_FILE, "w") as f:
        json.dump(c, f, indent=2)

def reddit_get(path, params=None, max_retries=3):
    """GET Reddit data. Uses OAuth if available, otherwise public JSON with fallback."""
    global _reddit_token, _reddit_use_oauth

    for attempt in range(max_retries):
        try:
            if _reddit_use_oauth:
                resp = requests.get(
                    f"https://oauth.reddit.com{path}",
                    headers={
                        "Authorization": f"bearer {_reddit_token}",
                        "User-Agent": REDDIT_USER_AGENT,
                    },
                    params=params, timeout=15,
                )
                if resp.status_code == 401:
                    log.warning("Reddit token expired, re-acquiring...")
                    if _try_reddit_oauth():
                        continue
                    _reddit_use_oauth = False  # degrade to public
            if not _reddit_use_oauth:
                # Public JSON fallback — try www then old.reddit.com
                # Build the URL: path like "/r/emacs/top" + ".json"
                json_path = path if path.endswith(".json") else path + ".json"
                for base in ["https://www.reddit.com", "https://old.reddit.com"]:
                    resp = requests.get(
                        f"{base}{json_path}",
                        headers={"User-Agent": REDDIT_USER_AGENT},
                        params=params, timeout=15,
                    )
                    if resp.ok:
                        break

            if resp.status_code == 429:
                wait = (2 ** attempt)
                log.warning(f"Reddit 429 — backing off {wait}s (attempt {attempt+1})")
                time.sleep(wait)
                continue
            resp.raise_for_status()
            return resp.json()
        except requests.RequestException as e:
            wait = (2 ** attempt)
            log.warning(f"Reddit request failed: {e} — retry in {wait}s")
            time.sleep(wait)

    log.error(f"Reddit GET {path} failed after {max_retries} retries.")
    return None

# ── LLM helpers ────────────────────────────────────────────────────────────
def validate_json_response(raw_text, required_fields):
    """Parse JSON and check for required fields.
    Returns (dict, None) on success or (None, error_str) on failure."""
    try:
        data = json.loads(raw_text)
    except json.JSONDecodeError as e:
        return None, f"Invalid JSON: {e} | Raw: {raw_text[:200]}"
    missing = [f for f in required_fields if f not in data]
    if missing:
        return None, f"Missing fields {missing} in: {raw_text[:200]}"
    return data, None

def llm_call(model, system_prompt, user_content):
    """Wrapper around ollama.chat that returns (parsed_dict, raw_text) or raises."""
    response = ollama.chat(
        model=model,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user",   "content": user_content},
        ],
        format="json",
        options={"temperature": 0.0},
    )
    raw = response["message"]["content"]
    return raw

# ── URL normalization for cross-source dedup ───────────────────────────────
def normalize_url(url):
    """Normalize URL for dedup comparison."""
    parsed = urlparse(url)
    host = (parsed.hostname or "").removeprefix("www.")
    path = parsed.path.rstrip("/")
    return f"{host}{path}".lower()

significant_commits = []

def get_filtered_commits(repo_path, days):
    subprocess.run(
        ["git", "-C", repo_path, "fetch", "origin", "+master:master"],
        capture_output=True, check=True,
    )
    git_cmd = [
        "git", "-C", repo_path, "log",
        f"--since={days} days ago",
        "--pretty=format:COMMIT_START%n%h|%an|%cs|%s%n%b",
        "--no-merges", "master",
    ]
    result = subprocess.run(git_cmd, capture_output=True, text=True, check=True)
    if "COMMIT_START" not in result.stdout:
        return []

    raw_commits = result.stdout.split("COMMIT_START\n")[1:]
    ignore_patterns = [
        r"^Bump version", r"^Fix typo", r"^; ", r"copyright",
        r"^Merge", r"^Auto-commit", r"^Update.*files",
        r"^admin", r"^Regenerate", r"^Add.*test", r"^[Tt]ests?:",
    ]
    filtered = []
    for raw in raw_commits:
        lines = raw.strip().split('\n')
        if not lines or len(lines[0].split('|')) < 4:
            continue
        meta    = lines[0].split('|')
        subject = meta[3]
        if any(re.search(p, subject, re.IGNORECASE) for p in ignore_patterns):
            continue
        filtered.append({
            "hash":    meta[0],
            "author":  meta[1],
            "date":    meta[2],
            "subject": subject,
            "body":    '\n'.join(lines[1:]).strip(),
        })
    return filtered

commit_system_prompt = """\
You are a strict technical evaluator for GNU Emacs development.

Review the commit message below. Classify it by these exact criteria:
  "Feature"      — a net-new, user-visible capability (not a bug fix or tweak)
  "Architecture" — a structural change to the C core, runtime, or a major subsystem
  "Trivial"      — anything else (fixes, tests, docs, minor refactors)

Also assign a confidence score 1–5 for how significant this change is to an
experienced Emacs user. Reserve 4–5 for changes that visibly affect daily
workflow or core infrastructure. Score 3 for solid but narrower improvements
(e.g. mode-specific features, useful fixes). Score 1–2 for trivial or internal.

Respond ONLY with a JSON object matching this exact schema:
{"is_significant": boolean, "category": "Feature" | "Architecture" | "Trivial", "confidence": integer, "reason": "One sentence."}
"""

log.info("Starting Git Triage...")
today_str    = datetime.date.today().isoformat()
all_commits  = get_filtered_commits(REPO_PATH, DAYS_AGO)
new_commits  = [c for c in all_commits if c["hash"] not in cache["commits"]]
log.info(f"  {len(all_commits)} commits after noise filter, {len(new_commits)} not yet seen.")

for i, commit in enumerate(new_commits):
    try:
        raw = llm_call(
            MODEL_TRIAGE,
            commit_system_prompt,
            f"Subject: {commit['subject']}\nBody:\n{commit['body']}",
        )
        result, err = validate_json_response(raw, ["is_significant", "category", "confidence", "reason"])
        if err:
            log.warning(f"  Validation failed for {commit['hash']}: {err}")
            continue
        cache["commits"][commit["hash"]] = today_str
        if result["is_significant"] and int(result.get("confidence", 0)) >= 4:
            commit["llm_eval"] = result
            significant_commits.append(commit)
    except Exception as e:
        log.error(f"  LLM call failed for {commit['hash']}: {e}")
    # Incremental cache save every 25 commits
    if (i + 1) % 25 == 0:
        save_cache(cache)
        log.debug(f"  Cache checkpoint at commit {i+1}/{len(new_commits)}")

save_cache(cache)
log.info(f"Git Triage Complete: {len(significant_commits)} high-confidence commits (>= 4/5).")

significant_posts = []

def fetch_reddit_emacs(limit=40):
    data = reddit_get("/r/emacs/top", params={"t": "month", "limit": limit})
    if data and "data" in data:
        return data["data"]["children"]
    log.warning("Failed to fetch r/emacs top posts.")
    return []

def fetch_top_comments(permalink, limit=5):
    # permalink looks like /r/emacs/comments/.../
    data = reddit_get(f"{permalink}.json")
    if not data or not isinstance(data, list) or len(data) < 2:
        return []
    children = data[1]["data"]["children"]
    return [c["data"].get("body", "") for c in children[:limit] if c["kind"] == "t1"]

def extract_quote_candidates(selftext, comments, min_len=50, max_len=220):
    raw = selftext + "\n" + "\n".join(comments)
    candidates = []
    for line in re.split(r'(?<=[.!?])\s+|\n', raw):
        line = line.strip().strip('"\'')
        if min_len <= len(line) <= max_len and not re.match(r'^(https?://|>|#|\*|\d+\.)', line):
            candidates.append(line)
    return candidates[:8]

def extract_links_from_html(html):
    """Extract (url, title, description) from <a> tags in Sacha's weekly HTML.
    Aggressively filters noise: bare URLs as titles, meetup/event links,
    social handles, and entries with no meaningful description."""
    links = []
    pattern = r'<a[^>]+href="([^"]+)"[^>]*>([^<]+)</a>([^<]*)'

    # Domains/paths that are navigation, social, or event noise
    skip_url_patterns = [
        'sachachua.com', 'twitter.com', 'x.com', 'youtube.com/channel',
        'meetup.com', 'emacslife.com/calendar', '.ics',
    ]
    # Titles that are just social handles, bare labels, or site names
    skip_title_patterns = [
        r'^https?://',            # title is a raw URL
        r'^@\S+',                 # Mastodon/Bluesky handle
        r'^Reddit$',              # bare "Reddit" back-link
        r'^Protesilaos$',         # bare author name as link text
        r'^Irreal$',
        r'^Hacker News$',
    ]

    for m in re.finditer(pattern, html):
        url   = m.group(1)
        title = unescape(m.group(2).strip())
        desc  = unescape(m.group(3).strip().strip('-–—:;,').strip())

        # Hard filters on URL and title length
        if len(title) < 8:
            continue
        if any(s in url for s in skip_url_patterns):
            continue
        if any(re.match(p, title) for p in skip_title_patterns):
            continue
        # Skip Mastodon/Bluesky/fediverse profile links
        if any(s in url for s in ['fosstodon.org', 'social.', 'bsky.app',
                                   'mastodon.', 'mathstodon.', 'fediscience.']):
            continue
        # Require a meaningful description (>10 chars) or a clearly descriptive title (>30 chars)
        if len(desc) < 10 and len(title) < 30:
            continue

        links.append({"url": url, "title": title, "description": desc})
    return links

def deduplicate_sources(posts, hn, lobsters, blogs, sacha):
    """Remove items appearing in multiple sources, keeping the richest version.
    Priority: Reddit > HN > Lobsters > Planet > Sacha."""
    seen = set()

    # Reddit posts — highest priority, keep all
    deduped_posts = []
    for p in posts:
        key = normalize_url(p["url"])
        seen.add(key)
        deduped_posts.append(p)

    # HN posts — drop if already covered by Reddit
    deduped_hn = []
    for h in hn:
        key = normalize_url(h["url"])
        if key not in seen:
            seen.add(key)
            deduped_hn.append(h)
        else:
            log.debug(f"  Dedup: HN '{h['title'][:50]}' already in Reddit")

    # Lobsters posts — drop if already covered
    deduped_lob = []
    for l in lobsters:
        key = normalize_url(l["url"])
        if key not in seen:
            seen.add(key)
            deduped_lob.append(l)
        else:
            log.debug(f"  Dedup: Lobsters '{l['title'][:50]}' already covered")

    # Planet blogs — drop if URL already covered
    deduped_blogs = []
    for b in blogs:
        key = normalize_url(b["url"])
        if key not in seen:
            seen.add(key)
            deduped_blogs.append(b)
        else:
            log.debug(f"  Dedup: blog '{b['title'][:50]}' already covered")

    # Sacha items — drop if URL already covered
    deduped_sacha = []
    for s in sacha:
        key = normalize_url(s["url"])
        if key not in seen:
            seen.add(key)
            deduped_sacha.append(s)

    return deduped_posts, deduped_hn, deduped_lob, deduped_blogs, deduped_sacha

def fetch_tips_and_tricks():
    data = reddit_get("/r/emacs/search", params={
        "q":           'title:"Fortnightly tips, tricks, & snippets"',
        "restrict_sr": "on",
        "sort":        "new",
        "limit":       "2",
    })
    if not data or "data" not in data:
        log.warning("Tips & Tricks search failed.")
        return []

    children = data["data"]["children"]
    if not children:
        return []

    # Use the permalink of the most recent tips thread
    latest_thread = children[0]["data"]["permalink"]

    # Reuse the fetch_top_comments function from Section 4
    return fetch_top_comments(latest_thread, limit=20)

log.info("Fetching Tips and Tricks...")
tips = fetch_tips_and_tricks()
fortune_path = os.path.join(OUTPUT_DIR, "tips_and_tricks.fortune")

if tips:
    # Load existing tips to avoid duplicates
    existing_tips = set()
    if os.path.exists(fortune_path):
        with open(fortune_path, "r") as f:
            for block in f.read().split("\n%\n"):
                existing_tips.add(block.strip())

    with open(fortune_path, "a") as f:
        added = 0
        for tip in tips:
            tip_text = tip.strip()
            if len(tip_text) > 50 and tip_text not in existing_tips:
                f.write(tip_text + "\n%\n")
                added += 1
    log.info(f"Appended {added} new tips to {fortune_path}")
    if added:
        log.info(f"Run 'strfile {fortune_path}' to compile for the fortune utility.")
else:
    log.info("No new tips found.")
